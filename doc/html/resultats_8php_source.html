221function marquerMinimums(stations) {
222 var mins = {};
223 stations.forEach(function(s) {
224 for (var c in s.prix) {
225 if (!(c in mins) || s.prix[c] < mins[c]) mins[c] = s.prix[c];
226 }
227 });
228 return stations.map(function(s) {
229 var estMin = {};
230 for (var c in s.prix) { estMin[c] = (s.prix[c] === mins[c]); }
231 s.estMin = estMin;
232 return s;
233 });
234}
