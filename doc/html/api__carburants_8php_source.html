118function normaliser_stations(array $records, string $carburant = ''): array
119{
120 $stations = [];

122 // Correspondance entre les noms de carburant affichés et les champs API
123 $champs_carburant = [
124 'SP95' => 'prix_sp95',
125 'SP98' => 'prix_sp98',
126 'E10' => 'prix_e10',
127 'E85' => 'prix_e85',
128 'Gazole' => 'prix_gazole',
129 'GPLc' => 'prix_gplc',
130 ];

132 foreach ($records as $rec) {
133 if (!is_array($rec)) {
134 continue;

137 // Extraction des prix disponibles
138 $prix = [];
139 foreach ($champs_carburant as $nom_affiche => $champ_api) {
140 $valeur = $rec[$champ_api] ?? null;
141 if ($valeur !== null && $valeur !== '' && (float)$valeur > 0) {
142 $prix[$nom_affiche] = (float)$valeur;
143 }
144 }
145
146 // Filtrage par carburant si précisé
147 if (!empty($carburant) && !isset($prix[$carburant])) {

152 if (empty($prix)) {
153 continue;
154 }

156 // Extraction des services (champ "services_service" = chaîne séparée par "//")
157 $services = [];
158 if (!empty($rec['services_service'])) {
159 $services = array_filter(
160 array_map('trim', explode('//', (string)$rec['services_service']))
161 );
162 $services = array_values($services);

191function marquer_prix_minimum(array $stations): array
192{
193 // 1. Calculer les prix minimums par carburant
194 $minimums = [];
195 foreach ($stations as $station) {
196 foreach ($station['prix'] as $carb => $prix) {
197 if (!isset($minimums[$carb]) || $prix < $minimums[$carb]) {
198 $minimums[$carb] = $prix;
199 }
200 }
201 }
202
203 // 2. Marquer chaque station
204 foreach ($stations as &$station) {
205 foreach ($station['prix'] as $carb => $prix) {
206 $station['est_minimum'][$carb] = isset($minimums[$carb]) && $prix === $minimums[$carb];
207 }
208 }
209 unset($station); // Bonne pratique : déréférencer la variable de boucle
210
211 return $stations;
